public class Customer {
    private int money;
    private boolean hasCoffee;

    public Customer() {
        money = 0;
        hasCoffee = false;
    }

    public Customer (int money, boolean hasCoffee) {
        this.money = money;
        this.hasCoffee = hasCoffee;
    }

    public void print() {
        System.out.println(money);
        System.out.println(hasCoffee);
    }

    //잔액이 부족할 경우
    public int pay(int coffeePrice) {
        if(money < coffeePrice) {
            System.out.println("[손님]의 작액이 부족합니다.");
            return -9;
        }
        money = money - coffeePrice;
        return 0;
    }

    public void order(Kiosk kiosk, Barista barista, String menuName) {
        System.out.println("[손님]이" + menuName + " 한 잔 주세요");

        int price = kiosk.getCoffeePrice();
        System.out.println("[바리스타] 커피 가격은 " + price + "원 입니다." );

        int paymentCheck = pay(price);
        if ( paymentCheck == -9) {
            return;
        }
        else if (paymentCheck == 0) {
            System.out.println("[손님]" + price + "원을 지불했습니다.");
            String coffee = barista.receiveOrder(menuName,price);
            takeCoffee(coffee);
        }
    }


    public void takeCoffee(String coffee) {
        hasCoffee = true;
    }
}
